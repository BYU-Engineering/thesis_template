% Full-page landscape figure (no float wrapper; reliable inside pdflscape)
% Usage: \landscapefigurepage[image height|large|slight]{image}{LOF caption}{figure caption}{label}
\newcommand{\landscapefigurepage}[5][0.78\textheight]{%
    \def\landscapefigopt{#1}%
    \def\landscapefiglarge{large}%
    \def\landscapefigslight{slight}%
    \def\landscapefigheight{#1}%
    \ifx\landscapefigopt\landscapefiglarge\def\landscapefigheight{0.80\textheight}\fi
    \ifx\landscapefigopt\landscapefigslight\def\landscapefigheight{0.80\textheight}\fi
    \begin{landscape}%
    \centering
    \includegraphics[width=\linewidth,height=\landscapefigheight,keepaspectratio]{#2}%
    \par\vspace{0.75em}%
    \captionof{figure}[#3]{#4}%
    \label{#5}%
    \end{landscape}%
}

% Appendix landscape diagram: section title + figure on one page (inside pdflscape)
% Usage: \landscapeappendixdiagram[image height]{section title}{section label}{image}{LOF caption}{figure caption}{figure label}
\newcommand{\landscapeappendixdiagram}[7][0.74\textheight]{%
    \section{#2}%
    \label{#3}%
    \par\vspace{0.25em}%
    \noindent\begin{minipage}{\linewidth}%
    \centering
    \includegraphics[width=\linewidth,height=#1,keepaspectratio]{#4}%
    \par\vspace{0.35em}%
    \captionof{figure}[#5]{#6}%
    \label{#7}%
    \end{minipage}%
}

% Portrait UI figures with lead-in text (figure title block) + image + caption.
% centeredfigurebottom — Fig. 5.2 at bottom of preceding text page (float [b]).
% centeredfigurepage — new page, block vertically centered (Figs. 5.3–5.8).
% centeredfigurepairpage — one page, two stacked centered figures (Figs. 5.9–5.10).
\makeatletter
\newcommand{\centeredfigure@block}[6]{%
  \begin{minipage}{0.9\textwidth}%
    \centering
    #2%
    \par\vspace{0.75em}%
    \includegraphics[width=#1,keepaspectratio]{#3}%
    \par\vspace{0.35em}%
    \captionof{figure}[#4]{#5}%
    \label{#6}%
  \end{minipage}%
}
\newcommand{\centeredfigurebottom}[6][0.80\textwidth]{%
  \par\vfil
  \noindent\centerline{\centeredfigure@block{#1}{#2}{#3}{#4}{#5}{#6}}%
  \par\vspace{0.5\baselineskip}%
}
\newcommand{\centeredfigurepage}[6][0.86\textwidth]{%
  \clearpage
  \null
  \vbox to \dimexpr\textheight-\baselineskip\relax{%
    \vfil
    \noindent\centerline{\centeredfigure@block{#1}{#2}{#3}{#4}{#5}{#6}}%
    \vfil
  }%
}
\newenvironment{centeredfigurepairpage}{%
  \clearpage
  \null
  \vbox to \dimexpr\textheight-\baselineskip\relax\bgroup
  \vfil
  \noindent\begin{minipage}{\textwidth}\centering
}{%
  \end{minipage}%
  \vfil\egroup
}
\newcommand{\centeredfigureblock}{\centeredfigure@block}
\makeatother
